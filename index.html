async function callGeminiAPI(apiKey, text, attachedFiles, promptStr, retries = 3, delay = 1000) { const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`; const parts = []; if (text) parts.push({ text: text }); attachedFiles.forEach(fileData => { parts.push({ inlineData: { mimeType: fileData.mimeType, data: fileData.data } }); }); // 翻訳以外の用途でも使えるようにJSON指示を調整 const jsonInstruction = `\n\nYOU MUST RESPOND ONLY WITH A VALID JSON ARRAY. No explanations, no markdown blocks outside the JSON. Format: [ { "speaker": "Speaker Name (or empty string. Use if applicable)", "sourceText": "Original text segment, user query, or topic description", "translatedText": "Translated text, answer, or your main response" } ] If the user asks a general question, put the context/question in 'sourceText' and your answer in 'translatedText'.`; const payload = { contents: [{ parts: parts }], systemInstruction: { parts: [{ text: promptStr + jsonInstruction }] }, generationConfig: { temperature: 0.2, responseMimeType: "application/json" }, // 【追加】セーフティフィルターを最も緩い設定（ブロックしない）に変更 safetySettings: [ { category: "HARM_CATEGORY_HARASSMENT", threshold: "BLOCK_NONE" }, { category: "HARM_CATEGORY_HATE_SPEECH", threshold: "BLOCK_NONE" }, { category: "HARM_CATEGORY_SEXUALLY_EXPLICIT", threshold: "BLOCK_NONE" }, { category: "HARM_CATEGORY_DANGEROUS_CONTENT", threshold: "BLOCK_NONE" } ] }; for (let i = 0; i < retries; i++) { try { const response = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (!response.ok) { if (response.status === 400) throw new Error("APIエラー (400): リクエスト形式が無効です。"); if (response.status === 403) throw new Error("APIエラー (403): APIキーが無効か権限がありません。"); if (response.status === 404) throw new Error("APIエラー (404): モデルエンドポイントが見つかりません。"); throw new Error(`APIエラー: ${response.status}`); } const result = await response.json(); let rawText = result.candidates?.[0]?.content?.parts?.[0]?.text; if (rawText) { rawText = rawText.replace(/```json/g, '').replace(/```/g, '').trim(); try { const parsed = JSON.parse(rawText); if (Array.isArray(parsed)) { return parsed; } else { return [parsed]; } } catch (parseError) { console.warn("JSON Parse Error, falling back to raw text rendering", parseError); return [{ speaker: "", sourceText: "", translatedText: rawText }]; } } else { // 【追加】エラーの詳細な理由をコンソールに出力し、画面にも表示する console.error("Full API Response:", result); const finishReason = result.candidates?.[0]?.finishReason; if (finishReason === 'SAFETY') { throw new Error("APIエラー: 不適切と判定されたため、Geminiのセーフティフィルターによって回答がブロックされました。"); } else if (finishReason === 'RECITATION') { throw new Error("APIエラー: 著作権保護（既存の文章との一致）により回答がブロックされました。"); } else { throw new Error("APIレスポンスにテキストが含まれていません。（詳細はブラウザの開発者ツールコンソールをご確認ください）"); } } } catch (error) { if (i === retries - 1) throw error; await new Promise(res => setTimeout(res, delay)); delay *= 2; } } }
